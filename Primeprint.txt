DECLARE
    n int:=&n;
    flag int;
BEGIN
    DBMS_OUTPUT.PUT_LINE('Prime numbers from 1 to ' || n || ':');
    FOR num IN 2..n LOOP
        flag := 1;
        FOR i IN 2..num-1 LOOP
            IF MOD(num, i) = 0 THEN
                flag := 0;
                EXIT;
            END IF;
        END LOOP;
        IF flag = 1 THEN
            DBMS_OUTPUT.PUT_LINE(num);
        END IF;
    END LOOP;
END;
/
